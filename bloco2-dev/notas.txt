# FASE 2 — Assembly (gcc -S)
# Cole aqui 5 a 10 linhas de soma.s que você considera mais
# interessantes e explique o que cada uma faz.

Trecho de soma.s:

main:
	pushq	%rbp
	movq	%rsp, %rbp
	movl	$30, %esi
	leaq	.LC0(%rip), %rax
	movq	%rax, %rdi
	call	printf@PLT
	popq	%rbp
	ret

Explicação:
Essas instruções montam a função main em assembly.
Elas organizam a pilha, carregam os argumentos do printf,
chamam a função printf e depois encerram a execução.
